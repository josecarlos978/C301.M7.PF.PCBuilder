## 📊 Estructura de Tablas SQL Recomendada

```sql
-- Tabla: categorias
CREATE TABLE categorias (
  id SERIAL PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL UNIQUE,
  descripcion TEXT
);

-- Tabla: subcategorias
CREATE TABLE subcategorias (
  id SERIAL PRIMARY KEY,
  categoria_id INTEGER NOT NULL REFERENCES categorias(id),
  nombre VARCHAR(100) NOT NULL,
  descripcion TEXT,
  UNIQUE(categoria_id, nombre)
);

-- Tabla: productos
CREATE TABLE productos (
  id SERIAL PRIMARY KEY,
  codigo_sku VARCHAR(50) NOT NULL UNIQUE,
  nombre VARCHAR(200) NOT NULL,
  subcategoria_id INTEGER NOT NULL REFERENCES subcategorias(id),
  marca VARCHAR(100),
  precio_venta DECIMAL(10, 2) NOT NULL,
  estado VARCHAR(20) DEFAULT 'activo',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Tabla: atributos (tipos de atributos dinámicos)
CREATE TABLE tipos_atributo (
  id SERIAL PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL UNIQUE,
  tipo_dato VARCHAR(50), -- texto, numero, booleano, etc
  descripcion TEXT
);

-- Tabla: producto_atributo (relación muchos a muchos)
CREATE TABLE producto_atributo (
  id SERIAL PRIMARY KEY,
  producto_id INTEGER NOT NULL REFERENCES productos(id),
  tipo_atributo_id INTEGER NOT NULL REFERENCES tipos_atributo(id),
  valor TEXT NOT NULL
);

-- Crear índices
CREATE INDEX idx_subcategoria_categoria ON subcategorias(categoria_id);
CREATE INDEX idx_producto_subcategoria ON productos(subcategoria_id);
CREATE INDEX idx_producto_atributo_producto ON producto_atributo(producto_id);
CREATE INDEX idx_producto_atributo_tipo ON producto_atributo(tipo_atributo_id);
```
